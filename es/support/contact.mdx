  </Card>

  </Card>
  <Card title="Reclamación de Paquetes" icon="file-circle-exclamation" href="/es/parcels/claims">

  <Card title="Reclamación de Palets" icon="file-circle-exclamation" href="/es/pallets/claims">

  </Card>
</CardGroup>
